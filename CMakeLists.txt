cmake_minimum_required(VERSION 3.22)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(ShareStream VERSION 0.5.0 LANGUAGES C CXX)
option(SHARE_STREAM_BUILD_PRODUCTS "Build distributable plugins, receiver and installer" ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_DRED OFF CACHE BOOL "" FORCE)
set(OPUS_OSCE OFF CACHE BOOL "" FORCE)
if(MSVC)
  set(OPUS_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
endif()
# libopus 1.6.1; the immutable upstream commit is shared by all native builds.
FetchContent_Declare(opus GIT_REPOSITORY https://github.com/xiph/opus.git
  GIT_TAG 22244de5a79bd1d6d623c32e72bf1954b56235be)
FetchContent_MakeAvailable(opus)
set_target_properties(opus PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(monocypher STATIC vendor/monocypher/monocypher.c)
target_include_directories(monocypher PUBLIC vendor/monocypher)
set_target_properties(monocypher PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(stream_core STATIC Source/Protocol.cpp Source/AudioCodec.cpp Source/StreamEngine.cpp Source/Invitation.cpp Source/ParticipantPresence.cpp Source/RoomDirectory.cpp Source/UpdateContract.cpp Source/Updater.cpp Source/InstallTransaction.cpp)
set_target_properties(stream_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(stream_core PUBLIC Source)
target_compile_definitions(stream_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=1 JUCE_MODAL_LOOPS_PERMITTED=0 SHARE_STREAM_VERSION="${PROJECT_VERSION}")
target_link_libraries(stream_core PUBLIC monocypher Opus::opus juce::juce_audio_basics juce::juce_core juce::juce_cryptography)
if(UNIX AND NOT APPLE)
  find_package(CURL REQUIRED)
  target_link_libraries(stream_core PUBLIC CURL::libcurl)
endif()
if(WIN32)
  target_link_libraries(stream_core PUBLIC ws2_32 bcrypt winmm)
endif()
if(SHARE_STREAM_BUILD_PRODUCTS)
set(plugin_formats VST3)
if(APPLE)
  list(APPEND plugin_formats AU)
endif()

juce_add_plugin(ShareStream
  COMPANY_NAME "Share Stream" BUNDLE_ID "audio.sharestream.plugin"
  PLUGIN_MANUFACTURER_CODE Shst PLUGIN_CODE Ss01
  FORMATS ${plugin_formats} PRODUCT_NAME "Share Stream"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  AU_MAIN_TYPE kAudioUnitType_Effect
  COPY_PLUGIN_AFTER_BUILD FALSE
  MICROPHONE_PERMISSION_ENABLED FALSE
  HARDENED_RUNTIME_ENABLED TRUE
  VST3_CATEGORIES Fx Network Generator)
target_sources(ShareStream PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(ShareStream PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ShareStream PRIVATE stream_core juce::juce_audio_utils juce::juce_recommended_config_flags)

juce_add_gui_app(ShareStreamReceiver PRODUCT_NAME "Share Stream"
  COMPANY_NAME "Share Stream" BUNDLE_ID "audio.sharestream.receiver"
  MICROPHONE_PERMISSION_ENABLED FALSE HARDENED_RUNTIME_ENABLED TRUE)
target_sources(ShareStreamReceiver PRIVATE Source/Standalone.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(ShareStreamReceiver PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ShareStreamReceiver PRIVATE stream_core juce::juce_audio_utils juce::juce_recommended_config_flags)
if(APPLE)
  set(native_notices "${CMAKE_CURRENT_BINARY_DIR}/bundle-resources/Licenses")
  file(MAKE_DIRECTORY "${native_notices}")
  configure_file(THIRD_PARTY.md "${native_notices}/THIRD_PARTY.md" COPYONLY)
  configure_file(vendor/Opus-COPYING "${native_notices}/Opus-COPYING" COPYONLY)
  configure_file(vendor/monocypher/LICENCE.md "${native_notices}/Monocypher-LICENCE.md" COPYONLY)
  foreach(native_target ShareStreamReceiver ShareStream_VST3 ShareStream_AU)
    juce_add_bundle_resources_directory(${native_target} "${native_notices}")
  endforeach()
endif()

endif() # SHARE_STREAM_BUILD_PRODUCTS

enable_testing()
add_executable(stream_tests Tests/StreamTests.cpp)
target_link_libraries(stream_tests PRIVATE stream_core)
target_compile_definitions(stream_tests PRIVATE SHARE_STREAM_INVITE_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/Tests/Invitations.json")
add_test(NAME stream_core COMMAND stream_tests)
add_executable(codec_tests Tests/CodecTests.cpp)
target_link_libraries(codec_tests PRIVATE stream_core)
add_test(NAME streaming_quality COMMAND codec_tests)
add_executable(stream_probe Tests/StreamProbe.cpp)
target_link_libraries(stream_probe PRIVATE stream_core)
add_executable(group_probe Tests/GroupProbe.cpp)
target_link_libraries(group_probe PRIVATE stream_core)

if(SHARE_STREAM_BUILD_PRODUCTS)
juce_add_console_app(plugin_probe PRODUCT_NAME "Share Stream Plugin Probe")
if(APPLE)
  target_sources(plugin_probe PRIVATE Tests/PluginProbe.mm)
  target_compile_definitions(plugin_probe PRIVATE JUCE_PLUGINHOST_AU=1)
else()
  target_sources(plugin_probe PRIVATE Tests/PluginProbe.cpp)
endif()
target_compile_definitions(plugin_probe PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=1)
target_link_libraries(plugin_probe PRIVATE stream_core juce::juce_audio_processors)
if(APPLE)
  target_link_libraries(plugin_probe PRIVATE "-framework CoreAudioKit")
  set_target_properties(plugin_probe PROPERTIES OSX_ARCHITECTURES "${CMAKE_HOST_SYSTEM_PROCESSOR}")
endif()
endif()
juce_add_console_app(processor_tests PRODUCT_NAME "Share Stream Processor Tests")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME group_transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/run_group_probe.py" "$<TARGET_FILE:group_probe>")
set_tests_properties(group_transport PROPERTIES TIMEOUT 90)
add_test(NAME stream_transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/run_stream_probe.py" "$<TARGET_FILE:stream_probe>")
set_tests_properties(stream_transport PROPERTIES TIMEOUT 240)
target_sources(processor_tests PRIVATE Tests/ProcessorTests.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(processor_tests PRIVATE Source)
target_link_libraries(processor_tests PRIVATE stream_core juce::juce_audio_utils)
target_compile_definitions(processor_tests PRIVATE SHARE_STREAM_RELAY_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/server/relay.py" SHARE_STREAM_PYTHON="${Python3_EXECUTABLE}")
target_compile_definitions(processor_tests PRIVATE SHARE_STREAM_MONITOR_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/Tests/MonitorGains.json")
if(SHARE_STREAM_BUILD_PRODUCTS)
  add_dependencies(processor_tests plugin_probe ShareStream_VST3)
  add_test(NAME plugin_factory COMMAND plugin_probe "${CMAKE_CURRENT_BINARY_DIR}/ShareStream_artefacts/Release/VST3/Share Stream.vst3")
endif()
if(APPLE AND SHARE_STREAM_BUILD_PRODUCTS)
  add_test(NAME processor_receive COMMAND processor_tests "$<TARGET_FILE:plugin_probe>" "${CMAKE_CURRENT_BINARY_DIR}/ShareStream_artefacts/Release/VST3/Share Stream.vst3")
else()
  add_test(NAME processor_receive COMMAND processor_tests)
endif()
set_tests_properties(processor_receive PROPERTIES TIMEOUT 60)
set_tests_properties(processor_receive PROPERTIES ENVIRONMENT "SHARE_STREAM_PRESENCE_URL=disabled")
if(APPLE)
  set_target_properties(processor_tests PROPERTIES OSX_ARCHITECTURES "${CMAKE_HOST_SYSTEM_PROCESSOR}")
endif()

add_executable(update_tests Tests/UpdateTests.cpp)
target_link_libraries(update_tests PRIVATE stream_core)
add_test(NAME updater_contract_and_install COMMAND update_tests)
add_executable(update_probe EXCLUDE_FROM_ALL Tests/UpdateProbe.cpp)
target_link_libraries(update_probe PRIVATE stream_core)

set(SHARE_STREAM_INSTALLER_PAYLOAD "" CACHE FILEPATH "Signed, stapled payload ZIP for the macOS installer")
if(SHARE_STREAM_BUILD_PRODUCTS AND APPLE AND SHARE_STREAM_INSTALLER_PAYLOAD)
  juce_add_gui_app(ShareStreamInstaller PRODUCT_NAME "Share Stream Setup"
    COMPANY_NAME "Share Stream" BUNDLE_ID "audio.sharestream.installer"
    MICROPHONE_PERMISSION_ENABLED FALSE HARDENED_RUNTIME_ENABLED TRUE)
  target_sources(ShareStreamInstaller PRIVATE Source/Installer.cpp)
  target_link_libraries(ShareStreamInstaller PRIVATE stream_core juce::juce_gui_extra)
  juce_add_bundle_resources_directory(ShareStreamInstaller "${native_notices}")
  set_property(TARGET ShareStreamInstaller APPEND PROPERTY LINK_DEPENDS "${SHARE_STREAM_INSTALLER_PAYLOAD}")
  add_custom_command(TARGET ShareStreamInstaller POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:ShareStreamInstaller>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SHARE_STREAM_INSTALLER_PAYLOAD}" "$<TARGET_BUNDLE_DIR:ShareStreamInstaller>/Contents/Resources/payload.zip")
endif()
